cmake_minimum_required(VERSION 3.20)
project(fielddb CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
add_compile_options(-Wall -Wextra)

option(FIELDDB_BUILD_TESTS "Build the engine tests and the regression tests" ON)
set(FIELDDB_ROWS 1500000 CACHE STRING "Rows loaded by the fielddb benchmark layer")
set(FIELDDB_CAP_MB 2048 CACHE STRING "Total log capacity of the fielddb layer, split across shards (MB)")

find_package(Threads REQUIRED)

# The modular engine (header-only), as developed in this repository's history.
add_library(field_engine INTERFACE)
target_include_directories(field_engine INTERFACE ${CMAKE_SOURCE_DIR}/storage)

# The single-header release used by the fielddb layer: dist/field_storage.hpp
# plus two downstream patches and the five fixes (see fielddb/README.md).
add_library(fielddb_release INTERFACE)
target_include_directories(fielddb_release INTERFACE ${CMAKE_SOURCE_DIR}/fielddb/include)

# The MVCC database layer used by the Postgres comparison (fielddb/bench/run.py
# builds its own copy per mode; this target is for manual runs).
add_executable(field_mvcc fielddb/db/field_mvcc.cpp)
target_compile_definitions(field_mvcc PRIVATE ROWS=${FIELDDB_ROWS} CAP_MB=${FIELDDB_CAP_MB})
target_link_libraries(field_mvcc PRIVATE fielddb_release Threads::Threads)

# Single-header distribution: dist/ is generated by tools/amalgamate.py.
add_executable(amalgam_smoke tools/amalgam_smoke.cpp)
target_include_directories(amalgam_smoke PRIVATE ${CMAKE_SOURCE_DIR}/dist)
add_executable(amalgam_smoke_builtins tools/amalgam_smoke.cpp)
target_include_directories(amalgam_smoke_builtins PRIVATE ${CMAKE_SOURCE_DIR}/dist)
target_compile_definitions(amalgam_smoke_builtins PRIVATE STORAGE_FORCE_ATOMIC_BUILTINS)

add_executable(platform_check embedded/platform_check.cpp)

if(FIELDDB_BUILD_TESTS)
    enable_testing()
    find_package(doctest REQUIRED)

    add_executable(engine_tests
        tests/doctest_main.cpp
        tests/test_storage.cpp tests/test_field_storage.cpp
        tests/test_indexes.cpp tests/test_refs_and_blobs.cpp)
    target_link_libraries(engine_tests PRIVATE field_engine doctest::doctest Threads::Threads)

    # Regression tests for the five engine fixes: they FAIL on the original
    # engine and pass on the release header.
    add_executable(fielddb_regress fielddb/tests/regress.cpp)
    target_link_libraries(fielddb_regress PRIVATE fielddb_release Threads::Threads)

    add_test(NAME engine_tests COMMAND engine_tests)
    add_test(NAME fielddb_regress COMMAND fielddb_regress)
    add_test(NAME amalgam_current COMMAND ${CMAKE_SOURCE_DIR}/tools/amalgamate.py --check)
    add_test(NAME amalgam_smoke COMMAND amalgam_smoke)
    add_test(NAME amalgam_smoke_builtins COMMAND amalgam_smoke_builtins)
    add_test(NAME platform_check COMMAND platform_check)
endif()
